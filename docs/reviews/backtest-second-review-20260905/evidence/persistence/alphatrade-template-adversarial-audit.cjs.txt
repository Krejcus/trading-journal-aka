const esbuild = require('/Users/filipkrejca/Documents/trading-journal-aka/node_modules/esbuild');
const vm = require('node:vm');
const assert = require('node:assert/strict');
const root = '/Users/filipkrejca/Documents/trading-journal-aka';
const id = '00000000-0000-4000-8000-000000000001';
const record = (patch = {}) => ({ id, indicator: 'levels', name: 'My levels', value: { color: 'local' }, updatedAt: '2026-09-01T02:00:00.000Z', ...patch });
const row = r => ({ id: r.id, indicator: r.indicator, name: r.name, value: r.deleted ? { __alphatradeChartTemplateDeleted: 1 } : r.value, updated_at: r.updatedAt });
const deferred = () => { let resolve; const promise = new Promise(done => { resolve = done; }); return { promise, resolve }; };
const tick = async () => { for (let i = 0; i < 8; i++) await Promise.resolve(); };

(async () => {
  const built = await esbuild.build({ entryPoints: [root + '/services/chartTemplateStore.ts'], bundle: true, format: 'cjs', platform: 'node', write: false,
    plugins: [{ name: 'mock-all-transport', setup(build) {
      build.onResolve({ filter: /^\.\/(supabase|storageService)$/ }, args => ({ path: args.path, namespace: 'mock' }));
      build.onLoad({ filter: /.*/, namespace: 'mock' }, args => ({ contents: args.path.endsWith('supabase') ? 'export const supabase = globalThis.__fake.supabase;' : 'export const getUserId = async () => globalThis.__fake.user;' }));
    } }],
  });
  function setup() {
    const storage = new Map();
    const fake = { user: 'A', auth: null, select: async () => ({ data: [], error: null }), writes: [], upsert: async value => { fake.writes.push(value); return { error: null }; } };
    fake.supabase = { auth: { onAuthStateChange(callback) { fake.auth = callback; } }, from() { return {
      select() { const filters = {}; const q = { eq(key, value) { filters[key] = value; return q; }, then(resolve, reject) { return fake.select(filters).then(resolve, reject); } }; return q; },
      upsert(value) { return fake.upsert(value); },
    }; } };
    const localStorage = { getItem: key => storage.get(key) ?? null, setItem: (key, value) => storage.set(key, value), removeItem: key => storage.delete(key) };
    const sandbox = { module: { exports: {} }, console: { error() {} }, setTimeout, clearTimeout, __fake: fake, window: { localStorage }, crypto: require('node:crypto').webcrypto };
    vm.runInNewContext(built.outputFiles[0].text, sandbox);
    return { api: sandbox.module.exports, fake, storage, localStorage };
  }
  const result = {};
  {
    const { api, fake, storage } = setup();
    const old = row(record({ value: { color: 'server-old' }, updatedAt: '2026-09-01T01:00:00.000Z' }));
    let server = old;
    storage.set(api.chartTemplateUserStorageKey('A'), JSON.stringify([record()]));
    fake.select = async () => {
      const snapshot = JSON.parse(JSON.stringify(server));
      server = row(record({ deleted: true, updatedAt: '2026-09-01T03:00:00.000Z' }));
      return { data: [snapshot], error: null };
    };
    fake.upsert = async value => { server = value; fake.writes.push(value); return { error: null }; };
    await api.syncChartTemplates();
    assert.equal(server.value.__alphatradeChartTemplateDeleted, undefined);
    assert.equal(server.value.color, 'local');
    result.concurrentDeletion = { newerRemoteDeleteOverwritten: true, returnedStatus: api.chartTemplateSyncStatus(), resultingTimestamp: server.updated_at };
  }
  {
    const { api, fake, storage, localStorage } = setup();
    storage.set(api.CHART_TEMPLATE_STORAGE_KEY, JSON.stringify([record()]));
    localStorage.setItem = () => { throw new Error('synthetic quota'); };
    const pending = deferred();
    fake.select = filters => filters.user_id === 'A' ? pending.promise : Promise.resolve({ data: [], error: null });
    const importing = api.importLegacyChartTemplates();
    await tick();
    assert.equal(api.chartTemplates().length, 1);
    fake.user = 'B'; fake.auth('SIGNED_IN', { user: { id: 'B' } });
    const syncingB = api.syncChartTemplates();
    pending.resolve({ data: null, error: { message: 'A offline' } });
    await Promise.all([importing, syncingB]);
    assert.equal(storage.has(api.CHART_TEMPLATE_STORAGE_KEY), false);
    assert.equal(storage.has(api.chartTemplateUserStorageKey('A')), false);
    assert.equal(fake.writes.length, 0);
    result.legacyImportAuthSwitch = { legacyCopyDeleted: true, userACachePresent: false, cloudWrites: 0, statusFromOtherUser: api.chartTemplateSyncStatus() };
  }
  {
    const { api, fake, storage } = setup();
    const accountVersion = record({ value: { color: 'account' } });
    storage.set(api.chartTemplateUserStorageKey('A'), JSON.stringify([accountVersion]));
    storage.set(api.CHART_TEMPLATE_STORAGE_KEY, JSON.stringify([record({ value: { color: 'legacy-unique-value' } })]));
    fake.select = async () => ({ data: [row(accountVersion)], error: null });
    await api.importLegacyChartTemplates();
    assert.equal(storage.has(api.CHART_TEMPLATE_STORAGE_KEY), false);
    assert.equal(api.chartTemplates()[0].value.color, 'account');
    result.legacyNameConflict = { conflictingLegacyValueDiscarded: true, survivingValue: 'account', conflictArchive: false };
  }
  console.log(JSON.stringify({ source: 'canonical chartTemplateStore, real module with mocked auth/storage/cloud, no network', ...result }, null, 2));
})().catch(error => { console.error(error); process.exitCode = 1; });
